## The Three Coordinate Spaces (documented here so nobody forgets)

```
     Space A                Space B                Space C
┌─────────────────┐    ┌─────────────────┐    ┌─────────────────┐
│ PHYSICAL PIXELS │    │ LOGICAL PIXELS  │    │  CLAUDE PIXELS  │
│  mss captures   │    │  Qt draws here  │    │ We send image   │
│  cursor API     │    │  (after DPI     │    │ resized to this │
│  returns here   │    │   division)     │    │ size            │
│                 │    │                 │    │                 │
│ e.g. 2880×1800  │    │ e.g. 1440×900   │    │ e.g. 1280×800   │
│ @ user (200%)   │    │ @ user          │    │ @ user          │
└────────┬────────┘    └────────▲────────┘    └────────┬────────┘
         │                      │                      │
         └──────────────────────┼──────────────────────┘
                                │
    physical = claude × scale_x, scale_y   ← capture.py does THIS
    logical  = physical / dpi_ratio        ← overlay.py does THIS
```

**capture.py lives in Space A ↔ Space C. It never touches Space B.**

from config import CANDIDATE_RESOLUTIONS


def pick_resolution(width: int, height: int) -> tuple[int, int]:
    if width <= 0 or height <= 0:
        raise ValueError(f"width and height must be positive, got ({width}, {height})")
    source_ratio = width / height
    best = min(
        CANDIDATE_RESOLUTIONS,
        key=lambda res: abs(source_ratio - (res[0] / res[1])),
    )
    return best
```

from PIL import Image


def resize_for_claude(
    img: Image.Image,
    target_w: int,
    target_h: int,
) -> tuple[Image.Image, float, float]:
    if target_w <= 0 or target_h <= 0:
        raise ValueError(f"target dims must be positive, got ({target_w}, {target_h})")
    scale_x = img.width / target_w
    scale_y = img.height / target_h
    resized = img.resize((target_w, target_h), Image.Resampling.LANCZOS)
    return resized, scale_x, scale_y
```

def unscale_claude_coords(
    claude_x: int,
    claude_y: int,
    scale_x: float,
    scale_y: float,
    monitor_left: int,
    monitor_top: int,
    target_w: int,
    target_h: int,
) -> tuple[int, int]:
    # Clamp to [0, target_w - 1] × [0, target_h - 1] before scaling.
    # Claude occasionally returns slightly out-of-bounds coords; Clicky
    # clamps in ElementLocationDetector.swift for the same reason.
    clamped_x = max(0, min(claude_x, target_w - 1))
    clamped_y = max(0, min(claude_y, target_h - 1))
    physical_x = monitor_left + int(clamped_x * scale_x)
    physical_y = monitor_top + int(clamped_y * scale_y)
    return physical_x, physical_y
```

import mss


def list_monitors() -> list[dict]:
    with mss.mss() as sct:
        # sct.monitors[0] is the virtual-desktop union; skip it.
        return [dict(m) for m in sct.monitors[1:]]


def monitor_containing(x: int, y: int, monitors: list[dict]) -> dict:
    for m in monitors:
        if (m["left"] <= x < m["left"] + m["width"]
                and m["top"] <= y < m["top"] + m["height"]):
            return m
    # Dead zone fallback: return primary (first in list).
    return monitors[0]
```

import ctypes
import ctypes.wintypes as wt


# Constants
_PROCESS_PER_MONITOR_DPI_AWARE_V2 = 2
_E_ACCESSDENIED = -2147024891  # 0x80070005


def set_dpi_awareness() -> bool:
    """Set per-monitor v2 DPI awareness. Idempotent."""
    try:
        hresult = ctypes.windll.shcore.SetProcessDpiAwareness(
            _PROCESS_PER_MONITOR_DPI_AWARE_V2
        )
        # SetProcessDpiAwareness returns S_OK (0) or E_ACCESSDENIED (-2147024891)
        # if already set. Treat either as success from our perspective.
        if hresult == 0:
            return True
        return False
    except OSError:
        # Already set by another library or OS doesn't support it.
        return False


class _POINT(ctypes.Structure):
    _fields_ = [("x", ctypes.c_long), ("y", ctypes.c_long)]


def get_cursor_position() -> tuple[int, int]:
    """Current cursor position in physical pixels (virtual desktop coords)."""
    pt = _POINT()
    ctypes.windll.user32.GetCursorPos(ctypes.byref(pt))
    return pt.x, pt.y
```

from dataclasses import dataclass


@dataclass
class CaptureResult:
    """All the metadata needed to send a screenshot to Claude and later
    map Claude's returned coordinates back to the user's physical screen."""
    image: Image.Image
    target_width: int
    target_height: int
    monitor: dict  # {'left', 'top', 'width', 'height'}
    cursor_physical: tuple[int, int]
    scale_x: float
    scale_y: float


def _capture_monitor(monitor: dict) -> Image.Image:
    """Grab a monitor's pixels via mss, return as PIL RGB image."""
    with mss.mss() as sct:
        sct_img = sct.grab(monitor)
        img = Image.frombytes("RGB", sct_img.size, sct_img.bgra, "raw", "BGRX")
        return img


def capture_active_screen() -> CaptureResult:
    set_dpi_awareness()  # Idempotent; safe to call every time.
    cursor_x, cursor_y = get_cursor_position()
    monitors = list_monitors()
    monitor = monitor_containing(cursor_x, cursor_y, monitors)
    raw_img = _capture_monitor(monitor)
    target_w, target_h = pick_resolution(raw_img.width, raw_img.height)
    resized, scale_x, scale_y = resize_for_claude(raw_img, target_w, target_h)
    return CaptureResult(
        image=resized,
        target_width=target_w,
        target_height=target_h,
        monitor=monitor,
        cursor_physical=(cursor_x, cursor_y),
        scale_x=scale_x,
        scale_y=scale_y,
    )
```
